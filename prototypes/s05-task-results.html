import pytest
from unittest.mock import patch, MagicMock
from auth.login import authenticate_user
from auth.exceptions import AuthenticationError, RateLimitError


class TestLoginFlow:
    """Unit tests for authentication login flow."""

    def test_login_valid_credentials(self, db_session):
        """Valid email/password returns access + refresh tokens."""
        result = authenticate_user(
            email="user@example.com",
            password="correct-password",
            session=db_session
        )
        assert result.access_token is not None
        assert result.refresh_token is not None
        assert result.expires_in == 3600

    def test_login_invalid_password(self, db_session):
        """Wrong password raises AuthenticationError."""
        with pytest.raises(AuthenticationError):
            authenticate_user(
                email="user@example.com",
                password="wrong-password",
                session=db_session
            )

    def test_login_nonexistent_user(self, db_session):
        """Non-existent email raises AuthenticationError."""
        with pytest.raises(AuthenticationError, match="User not found"):
            authenticate_user(
                email="nobody@example.com",
                password="any-password",
                session=db_session
            )

    def test_login_expired_token_refresh(self, db_session):
        """Expired access token triggers refresh flow."""
        result = authenticate_user(
            email="user@example.com",
            password="correct-password",
            session=db_session
        )
        # Simulate token expiry
        with patch("auth.token.is_expired", return_value=True):
            refreshed = result.refresh()
            assert refreshed.access_token != result.access_token
            assert refreshed.expires_in == 3600

    def test_login_rate_limit_exceeded(self, db_session):
        """5+ failed attempts in 60s raises RateLimitError."""
        for attempt in range(5):
            with pytest.raises(AuthenticationError):
                authenticate_user(
                    email="user@example.com",
                    password="wrong",
                    session=db_session
                )
        with pytest.raises(RateLimitError, match="Too many attempts"):
            authenticate_user(
                email="user@example.com",
                password="correct-password",
                session=db_session
            )

    def test_login_sql_injection_prevention(self, db_session):
        """SQL injection attempts are sanitized."""
        malicious_inputs = [
            "' OR '1'='1",
            "admin'--",
            "'; DROP TABLE users;--",
        ]
        for payload in malicious_inputs:
            with pytest.raises(AuthenticationError):
                authenticate_user(
                    email=payload,
                    password=payload,
                    session=db_session
                )
